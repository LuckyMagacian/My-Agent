@lexical/react

此包为 Lexical 提供了一套组件和 hooks，支持在 React 应用中进行文本编辑。

快速开始​

安装 lexical 和 @lexical/react：

                              
                                

                                  npm install --save lexical @lexical/react

以下是一个使用 lexical 和 @lexical/react 构建基础纯文本编辑器的示例（亲自尝试）。

                              
                                
import {$getRoot, $getSelection} from 'lexical';

                                
import {useEffect} from 'react';

                                


                                
import {LexicalComposer} from '@lexical/react/LexicalComposer';

                                
import {PlainTextPlugin} from '@lexical/react/LexicalPlainTextPlugin';

                                
import {ContentEditable} from '@lexical/react/LexicalContentEditable';

                                
import {HistoryPlugin} from '@lexical/react/LexicalHistoryPlugin';

                                
import {OnChangePlugin} from '@lexical/react/LexicalOnChangePlugin';

                                
import {useLexicalComposerContext} from '@lexical/react/LexicalComposerContext';

                                


                                
const theme = {

                                
  // 在这里设置主题样式

                                
  ...

                                
}

                                


                                
// 当 editor 发生变化时，你可以通过 OnChangePlugin 收到通知

                                
// LexicalOnChangePlugin!

                                
function onChange(editorState) {

                                
  editorState.read(() => {

                                
    // 在这里读取 EditorState 的内容

                                
    const root = $getRoot();

                                
    const selection = $getSelection();

                                


                                
    console.log(root, selection);

                                
  });

                                
}

                                


                                
// Lexical React plugins 是 React 组件，这使得它们

                                
// 具有高度的可组合性。此外，如果需要，你可以延迟加载 plugins，

                                
// 这样在你真正使用它们之前不会产生开销。

                                
function MyCustomAutoFocusPlugin() {

                                
  const [editor] = useLexicalComposerContext();

                                


                                
  useEffect(() => {

                                
    // 当 effect 触发时，让 editor 获得焦点

                                
    editor.focus();

                                
  }, [editor]);

                                


                                
  return null;

                                
}

                                


                                
// 捕获 Lexical 更新期间发生的任何错误并记录它们

                                
// 或根据需要抛出它们。如果你不抛出它们，Lexical 将

                                
// 尝试优雅地恢复，而不会丢失用户数据。

                                
function onError(error) {

                                
  throw error;

                                
}

                                


                                
function Editor() {

                                
  const initialConfig = {

                                
    namespace: 'MyEditor',

                                
    theme,

                                
    onError,

                                
  };

                                


                                
return (

                                
    <LexicalComposer initialConfig={initialConfig}>

                                
      <PlainTextPlugin

                                
        contentEditable={

                                
          <ContentEditable

                                
            aria-placeholder={'输入一些文本...'}

                                
            placeholder={<div>输入一些文本...</div>}

                                
          />

                                
        }

                                
      />

                                
      <OnChangePlugin onChange={onChange} />

                                
      <HistoryPlugin />

                                
      <MyCustomAutoFocusPlugin />

                                
    </LexicalComposer>

                                
  );

                                
}
